import { useEffect, useRef, useState } from 'react';
import type { JSX } from 'react';
import type { Message } from './presentational.tsx';
import { Presentational } from './presentational.tsx';

interface ChatWidgetProps {
  bffUrl: string;
}

const ChatWidget = ({ bffUrl }: ChatWidgetProps): JSX.Element => {
  const [isOpen, setIsOpen] = useState(false);
  const [messages, setMessages] = useState<Message[]>([]);
  const [inputValue, setInputValue] = useState('');
  const [isStreaming, setIsStreaming] = useState(false);
  const [sessionId] = useState<string>(() => crypto.randomUUID());
  const messagesEndRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const handleSend = (): void => {
    const trimmedInput = inputValue.trim();
    if (!trimmedInput || isStreaming) return;

    setInputValue('');
    setMessages((previous) => [
      ...previous,
      { role: 'user', content: trimmedInput },
      { role: 'assistant', content: '' },
    ]);
    setIsStreaming(true);

    const eventSource = new EventSource(`${bffUrl}/chat/${sessionId}`);

    const closeWithError = (message: string): void => {
      setMessages((previous) => {
        const updated = [...previous];
        const last = updated[updated.length - 1];
        if (last) last.content = message;
        return updated;
      });
      eventSource.close();
      setIsStreaming(false);
    };

    eventSource.addEventListener(
      'ready',
      () => {
        void fetch(`${bffUrl}/chat/${sessionId}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ inputMessage: trimmedInput }),
        }).catch(() => {
          closeWithError('Error: failed to send message');
        });
      },
      { once: true },
    );

    eventSource.addEventListener('output-message', (event) => {
      const parsed: unknown = JSON.parse(event.data as string);
      if (typeof parsed === 'object' && parsed !== null && 'chunk' in parsed) {
        setMessages((previous) => {
          const updated = [...previous];
          const last = updated[updated.length - 1];
          if (last)
            last.content += String((parsed as { chunk: unknown }).chunk);
          return updated;
        });
      }
    });

    eventSource.addEventListener('done', () => {
      eventSource.close();
      setIsStreaming(false);
    });

    eventSource.addEventListener('error', (event) => {
      const parsed: unknown = JSON.parse(
        (event as MessageEvent).data as string,
      );
      closeWithError(
        `Error: ${
          typeof parsed === 'object' && parsed !== null && 'error' in parsed
            ? String((parsed as { error: unknown }).error)
            : 'unknown error'
        }`,
      );
    });

    eventSource.onerror = () => {
      if (eventSource.readyState === EventSource.CLOSED) return;
      closeWithError('Error: connection failed');
    };
  };

  return (
    <Presentational
      isOpen={isOpen}
      messages={messages}
      inputValue={inputValue}
      isStreaming={isStreaming}
      messagesEndRef={messagesEndRef}
      onClose={() => {
        setIsOpen(false);
      }}
      onToggle={() => {
        setIsOpen((previous) => !previous);
      }}
      onInputChange={setInputValue}
      onSend={handleSend}
    />
  );
};

export default ChatWidget;
